def digest(path):
 h=hashlib.sha256()
 with open(path,'rb') as f:
  for block in iter(lambda:f.read(1048576), b''): h.update(block)
 return h.hexdigest()

def save():
 (work/'results.json').write_text(json.dumps(report,indent=2,sort_keys=True)+'\n')

def run(argv,cwd,log):
 log=pathlib.Path(log); log.parent.mkdir(parents=True,exist_ok=True)
 start=time.monotonic()
 with log.open('w') as f:
  result=subprocess.run([str(x) for x in argv],cwd=cwd,stdout=f,stderr=subprocess.STDOUT,env={**os.environ,'LEAN_NUM_THREADS':'2','CARGO_BUILD_JOBS':'2','NIX_BUILD_CORES':'2'})
 record={'argv':[str(x) for x in argv],'cwd':str(cwd),'log':str(log),'exit_code':result.returncode,'seconds':round(time.monotonic()-start,3)}
 report['commands'].append(record); save()
 return result.returncode

def need(argv,cwd,log):
 code=run(argv,cwd,log)
 if code: raise RuntimeError(f'command failed ({code}): {argv}; log {log}')

 def compile_case(name,replacements):
  case=work/name; case.mkdir()
  src=case/'zisk'; std=case/'proofman'
  need(['cp','-r','--reflink=never','--no-preserve=mode',str(zisk),str(src)],repo,case/'copy-zisk.log')
  need(['cp','-r','--reflink=never','--no-preserve=mode',str(proofman),str(std)],repo,case/'copy-proofman.log')
  original=(std/stdlib_rel).read_text(); text=original
  for before,after in replacements:
   if text.count(before)!=1: raise RuntimeError(f'{name}: source anchor count {text.count(before)} for {before!r}')
   text=text.replace(before,after)
  (std/stdlib_rel).write_text(text)
  wrapper_text=wrapper.read_text()
  old=str(proofman/'pil2-components/lib/std/pil')
  if wrapper_text.count(old)!=1: raise RuntimeError('production stdlib include anchor differs')
  wrapper_text=wrapper_text.replace(old,str(std/'pil2-components/lib/std/pil'))
  # Only the stdlib include and the cache-safe copy mode differ from production.
  wrapper_text=wrapper_text.replace('cp --no-preserve=mode','cp --reflink=never --no-preserve=mode')
  private_wrapper=case/'compile-source'; private_wrapper.write_text(wrapper_text); private_wrapper.chmod(0o755)
  pilout=case/'zisk.pilout'
  need([private_wrapper,src,pilout],repo,case/'compile.log')
  return case,pilout,{'stdlib_before_sha256':hashlib.sha256(original.encode()).hexdigest(),'stdlib_after_sha256':digest(std/stdlib_rel),'pilout_sha256':digest(pilout)}
